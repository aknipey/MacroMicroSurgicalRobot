\newglossaryentry{sym:q}{type=symbols,name={\ensuremath{\vec{q}, \qd}},sort=q,
  description={joint positions and velocities of the arm}}
\newglossaryentry{sym:J}{type=symbols,name={\ensuremath{\Jac}},sort=J,
  description={geometric Jacobian of the arm at \frm{tool0}}}
\newglossaryentry{sym:c}{type=symbols,name={\ensuremath{\cond}},sort=c,
  description={condition number of \Jac}}
\newglossaryentry{sym:xi}{type=symbols,name={\ensuremath{\twist = (\vlin, \vang)}},sort=xi,
  description={Cartesian twist sent to \servo}}
\newglossaryentry{sym:s}{type=symbols,name={\ensuremath{\scale}},sort=s,
  description={motion scale (1, 1/5 or 1/20)}}
\newglossaryentry{sym:kff}{type=symbols,name={\ensuremath{\kff}},sort=kff,
  description={velocity feedforward gain}}
\newglossaryentry{sym:Kp}{type=symbols,name={\ensuremath{\Kp, \Kr}},sort=Kp,
  description={translational and rotational correction gains}}
\newglossaryentry{sym:d}{type=symbols,name={\ensuremath{\depth}},sort=d,
  description={insertion depth of the tool tip past the RCM point}}
\newglossaryentry{sym:r}{type=symbols,name={\ensuremath{\lever}},sort=r,
  description={lever arm from \frm{tool0} to the RCM point at depth \depth}}

  description={RCM sideways gain: \frm{tool0} motion per unit of scaled stylus motion}}
