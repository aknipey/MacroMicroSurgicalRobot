% ---- Units: siunitx if installed (texlive-science), otherwise a fallback ----
% The fallback covers \SI, \si, \num, \qty, \unit, \ang with plain formatting so
% the document always builds; install texlive-science for the real thing.
\newif\ifhavesiunitx
\IfFileExists{siunitx.sty}{\havesiunitxtrue}{\havesiunitxfalse}
\ifhavesiunitx
  \usepackage{siunitx}
  \sisetup{detect-all, range-phrase={--}, range-units=single, per-mode=symbol,
           separate-uncertainty=true, group-minimum-digits=4}
\else
  \typeout{*** siunitx not found: using fallback unit macros ***}
  \newcommand{\num}[1]{#1}
  \newcommand{\si}[1]{\ensuremath{\mathrm{#1}}}
  \newcommand{\SI}[2]{#1\,\si{#2}}
  \newcommand{\qty}[2]{#1\,\si{#2}}
  \newcommand{\unit}[1]{\si{#1}}
  \newcommand{\ang}[1]{#1\ensuremath{^\circ}}
  \newcommand{\milli}{m}\newcommand{\micro}{\ensuremath{\mu}}
  \newcommand{\metre}{m}\newcommand{\second}{s}\newcommand{\hertz}{Hz}
  \newcommand{\radian}{rad}\newcommand{\degree}{\ensuremath{^\circ}}
  \newcommand{\newton}{N}\newcommand{\ampere}{A}\newcommand{\kilogram}{kg}
  \newcommand{\per}{/}\newcommand{\squared}{\ensuremath{^2}}\newcommand{\cubed}{\ensuremath{^3}}
\fi
